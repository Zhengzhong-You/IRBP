add_library(irbp_core STATIC src/problem.cpp src/path_search.cpp src/arc.cpp src/local_search.cpp src/pool_reduction.cpp)
target_include_directories(irbp_core PUBLIC include ${IRBP_THIRD_PARTY})
target_link_libraries(irbp_core PUBLIC routeopt_solver irbp_zip Threads::Threads)
if(MSVC)
  target_compile_definitions(irbp_core PUBLIC NOMINMAX)
  target_compile_options(irbp_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(irbp_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(irbp src/main.cpp)
target_link_libraries(irbp PRIVATE irbp_core)
if(WIN32)
  target_link_libraries(irbp PRIVATE psapi)
endif()
